# Optional OCSR stack for the hand-drawn-structure cross-check
# (chemextract/structures.py). Heavy + finicky pins, so it's kept out of the
# core requirements.txt. The name-resolution leg (rdkit + py2opsin) is light;
# MolScribe drags in an old OpenNMT-py / timm world that fights modern torch.
#
# Install order matters — MolScribe and OpenNMT-py must go in with --no-deps,
# because their pins (torch<2.0, pyonmttok, albumentations==1.1.0) either don't
# exist for current Python or conflict with the torch already installed:
#
#   pip install rdkit py2opsin                       # name leg (light)
#   pip install timm==0.4.12 SmilesPE albumentations==1.1.0 \
#               opencv-python-headless pandas scipy   # molscribe runtime deps
#   pip install --no-deps OpenNMT-py==2.2.0           # decoder modules only
#   pip install --no-deps molscribe                   # the OCSR model
#
# Notes:
# - Needs a JRE on PATH (py2opsin runs OPSIN, a Java jar).
# - MolScribe's checkpoint (~1 GB) is fetched from the HF Hub on first use.
# - structures.py shims the legacy torchtext API that OpenNMT-py imports but
#   MolScribe never executes, and runs MolScribe's post-processing Pool serially
#   so macOS 'spawn' workers don't re-import (and fail on) that shim.

rdkit>=2024.3
py2opsin>=1.2
timm==0.4.12
SmilesPE>=0.0.3
albumentations==1.1.0
opencv-python-headless>=4.5
pandas>=1.3
scipy>=1.7
# install these two with --no-deps (see above):
OpenNMT-py==2.2.0
molscribe>=1.1.1
